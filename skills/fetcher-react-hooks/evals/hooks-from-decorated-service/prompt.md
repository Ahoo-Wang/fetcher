Generate React hooks from my decorator-based UserService so components can call getUser and updateUser. Here it is:

```ts
@api('/users')
export class UserService {
  @get('/{id}')
  getUser(@path('id') id: string): Promise<User> {
    throw autoGeneratedError(id);
  }

  @put('/{id}')
  updateUser(@path('id') id: string, @body() user: UpdateUser): Promise<User> {
    throw autoGeneratedError(id, user);
  }
}

export const userService = new UserService();
```
